Register.Relation.radial_acceleration_relation : ∀ (R : Register.Reg) (G : ℝ),
  G = R.c ^ 3 * R.kB / (4 * R.ℏ * R.η) →
    ∀ (Λ : ℝ),
      0 < Λ →
        R.L * R.ℓc = 2 * R.pi * √(3 / Λ) →
          ∀ (TdS : ℝ),
            R.kB * TdS = R.ℏ / (R.L * R.tc) →
              ∀ (εs : ℝ),
                εs = R.kB * TdS →
                  ∀ (M r : ℝ),
                    0 < M →
                      0 < r →
                        ∀ (TN : ℝ),
                          R.kB * TN = R.ℏ / (R.La (G * M / r ^ 2) * R.tc) →
                            ∀ (a Ta : ℝ),
                              0 < a →
                                R.kB * Ta = R.ℏ / (R.La a * R.tc) →
                                  ∀ (surv : ℝ → ℝ),
                                    (∀ (x : ℝ), 0 ≤ x → surv x = Real.exp (-x)) →
                                      ∀ (b : ℝ),
                                        b = 1 - surv √(2 * R.kB * TN * R.ln2 / ((1 - R.ln2) * εs)) →
                                          M * R.c ^ 2 = 2 * (b * R.Ns r) * (R.kB * Ta * R.ln2) →
                                            a = G * M / r ^ 2 / (1 - Real.exp (-√(G * M / r ^ 2 / R.a0))) ∧
                                              R.a0 = (1 - R.ln2) * R.c ^ 2 / (2 * R.ln2 * √(3 / Λ)) ∧
                                                M * R.c ^ 2 = 2 * R.Ns r * (R.kB * TN * R.ln2)
